import { slate } from "@slate/sdk";
import { useQuery, useSource } from "@slate/sdk/react";
import { Badge, Card, Page, Row, Section } from "@slate/ui";
import { createRoot } from "react-dom/client";

function age(iso: string): string {
const h = Math.round((Date.now() - Date.parse(iso)) / 3_600_000);
return h < 24 ? h + "h" : Math.round(h / 24) + "d";
}

function App() {
const reviews = useSource(() => slate.sources.reviews());
const mine = useSource(() => slate.sources.pulls());
const todo = useQuery(() => slate.tasks.list({ section: "review", open: true }));
const waiting = reviews.rows.length;
return (
<Page
title="PR review queue"
description="Pull requests waiting on you, oldest first, and your own still open."
actions={<Badge tone={waiting ? "attention" : "ok"}>{waiting} to review</Badge>}
>
<Section title="Waiting for your review">
<Card padded={false}>
{reviews.rows.length === 0 && <Row>Nothing waiting on you.</Row>}
{[...reviews.rows]
.sort((a, b) => Date.parse(a.updatedAt) - Date.parse(b.updatedAt))
.map((pr) => (
<Row key={pr.repo + pr.number}>
<span className="w-16 shrink-0 text-muted">#{pr.number}</span>
<span className="min-w-0 flex-1 truncate">{pr.title}</span>
<span className="text-muted">{pr.author} · {age(pr.updatedAt)}</span>
</Row>
))}
</Card>
</Section>
<Section title="Yours, still open">
<Card padded={false}>
{mine.rows.map((pr) => (
<Row key={pr.repo + pr.number}>
<span className="w-16 shrink-0 text-muted">#{pr.number}</span>
<span className="min-w-0 flex-1 truncate">{pr.title}</span>
{pr.draft && <Badge>draft</Badge>}
</Row>
))}
</Card>
</Section>
<Section title="In Tasks">
<Card>{todo.data ? todo.data.length + " review items in Tasks" : "…"}</Card>
</Section>
</Page>
);
}

createRoot(document.getElementById("root") as HTMLElement).render(<App />);
